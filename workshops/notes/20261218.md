```
import pyrosetta

# 1. Rosetta 초기화
pyrosetta.init()

# 2. PDB 파일 불러오기
pose = pyrosetta.pose_from_pdb("단백질.pdb")
print(pose.total_residue())  # 잔기 개수를 본다
print(pose.sequence())  # 서열 한 줄을 본다

# 3. 에너지 함수 불러오기
# 같은 길이, 같은 단백질끼리 비교했을 때 숫자가 낮을수록 더 안정적인 구조.
scorefxn = pyrosetta.get_fa_scorefxn()
scorefxn = pyrosetta.get_score_function()
print(scorefxn(pose))
scorefxn.show(pose) 

# 4. 단백질 에너지 계산
energy = scorefxn(pose)

print("Number of residues:", pose.total_residue())
print("Total energy:", energy)
```

# FastRelax로 구조 최적화
`relax.apply(pose)`
- 주의할 점은 에너지가 낮아졌다고 반드시 생물학적으로 정확한 구조가 되었다는 뜻은 아님. 
- 입력 구조의 품질과 최적화 조건에 따라 결과가 달라질 수 있음.
- 

```
import pyrosetta
from pyrosetta.rosetta.protocols.relax import FastRelax

pyrosetta.init()

# PDB 구조 불러오기
pose = pyrosetta.pose_from_pdb("input.pdb")

# 에너지 함수 설정
scorefxn = pyrosetta.get_fa_scorefxn()

# 최적화 전 점수
before = scorefxn(pose)
print("Before:", before)

# FastRelax 실행
relax = FastRelax()
relax.set_scorefxn(scorefxn)
relax.apply(pose)

# 최적화 후 점수
after = scorefxn(pose)
print("After:", after)

# 결과 저장
pose.dump_pdb("relaxed.pdb")
```
