import numpy as np
import biotite.structure as struc
import biotite.structure.io as strucio

from biotite.structure import rmsd, superimpose
from atomworks.constants import PROTEIN_BACKBONE_ATOM_NAMES


af3_path = "res_AF3/fold_2026_11_24_10_54_model_0.cif"

aa_af3 = strucio.load_structure(af3_path, model=1)
aa_rf3 = rf3_outputs["test_5aa"][0].atom_array

# ==========================================
# 3. Filter backbone atoms (N, CA, C, O)
# ==========================================

bb_af3 = aa_af3[
    np.isin(aa_af3.atom_name, PROTEIN_BACKBONE_ATOM_NAMES)
]

bb_rf3 = aa_rf3[
    np.isin(aa_rf3.atom_name, PROTEIN_BACKBONE_ATOM_NAMES)
]

# ==========================================
# 4. Match corresponding atoms
# ==========================================

# Match by chain ID, residue ID, insertion code, atom name

def atom_key(atom):
    return (
        str(atom.chain_id),
        int(atom.res_id),
        str(atom.ins_code),
        str(atom.atom_name)
    )

rf3_atom_dict = {
    atom_key(atom): i
    for i, atom in enumerate(bb_rf3)
}

af3_indices = []
rf3_indices = []

for i, atom in enumerate(bb_af3):
    key = atom_key(atom)
    if key in rf3_atom_dict:
        af3_indices.append(i)
        rf3_indices.append(rf3_atom_dict[key])

bb_af3_matched = bb_af3[np.array(af3_indices)]
bb_rf3_matched = bb_rf3[np.array(rf3_indices)]

print("AF3 backbone atoms:", len(bb_af3))
print("RF3 backbone atoms:", len(bb_rf3))
print("Matched backbone atoms:", len(af3_indices))

if len(af3_indices) == 0:
    raise ValueError(
        "No matching backbone atoms found. "
        "Check chain IDs and residue numbering."
    )

# ==========================================
# 5. Superimpose and calculate RMSD
# ==========================================

bb_rf3_fitted, transform = superimpose(
    bb_af3_matched,
    bb_rf3_matched
)

rmsd_value = rmsd(
    bb_af3_matched,
    bb_rf3_fitted
)

print(f"\nAF3 vs RF3 Backbone RMSD: {rmsd_value:.2f} Å")

print(
    "Interpretation:",
    "Excellent agreement" if rmsd_value < 1.0
    else "Good agreement" if rmsd_value < 2.0
    else "Moderate or low agreement"
)
